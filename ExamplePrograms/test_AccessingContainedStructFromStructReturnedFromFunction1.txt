    fourth : int;

{
    a : int;

}

function main : int ()
{
    result : example2;
    result = coolFunction().tenth;

    yesOrNo : int;
    if (result.a == 10 && result.b == 11)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function coolFunction : example ()
{
    dep : example2;

    next.third = 3;

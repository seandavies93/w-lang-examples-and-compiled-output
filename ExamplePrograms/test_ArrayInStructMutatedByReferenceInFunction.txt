struct example
{
    integers : int{3};
    final : int;
}

function main : int ()
{
    var1 : example;

    var1.integers[0] = 1;
    var1.integers[1] = 2;
    var1.integers[2] = 3;
    var1.final = 4;

    mutate(var1.integers);

    yesOrNo : int;

    if (
        var1.integers[0] == 2 && 
        var1.integers[1] == 4 && 
        var1.integers[2] == 6 && 
        var1.final == 4)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function mutate : void (array : @int)
{
    array[0] = array[0] * 2;
    array[1] = array[1] * 2;
    array[2] = array[2] * 2;
}

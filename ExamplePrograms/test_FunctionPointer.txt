function main : int ()
{
    n : int;
    funcPointer : int[int];
    funcPointer = &factorial;
    n = funcPointer(5);

    yesOrNo : int;
    if (n == 120)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function factorial : int (n : int)
{
    value : int;
    value = 1;

    index : int;
    index = 1;
    
    while (index <= n)
    {
        value = value * index;
        index = index + 1;
    }

    return value;
}

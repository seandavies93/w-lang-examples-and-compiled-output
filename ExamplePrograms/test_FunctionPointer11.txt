struct example
{
    first : int;
    second : int;
    third : int;
}

struct example2 
{
    contained : example;
}

struct example3
{
    point : @example2;
    point1 : @example2;
}

function main : int ()
{
    var1 : example3;
    var2 : example2;
    
    var1.point1 = &var2;

    pointer : @example3;
    pointer = &var1;

    funcPointer : void[@example3];
    funcPointer = &setupPointer;

    funcPointer(pointer);
    
    resultStruct : example;
    result : int;
    yesOrNo : int;

    resultStruct = pointer->point1->contained;
    result = resultStruct.first;

    if (result == 1)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function setupPointer : void(pointer : @example3)
{
    funcPointer : example[];
    funcPointer = &create;
    pointer->point1->contained = funcPointer();
}

function create : example () 
{
    next : example;
    next.first = 1;
    next.second = 2;
    next.third = 3;
    return next;
}

struct example
{
    first : int;
    second : int;
    third : int;
}

struct example1
{
    funcPointer : example[];
}

function main : int ()
{
    retVal : example;
    funcPointerContainer : example1;
    funcPointerContainer.funcPointer = &coolFunction;
    retVal = funcPointerContainer.funcPointer();

    result : int;
    result = retVal.third;

    yesOrNo : int;
    if (result == 3)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function coolFunction : example ()
{
    next : example;
    next.first = 1;
    next.second = 2;
    next.third = 3;
    return next;
}

struct container
{
    funcPointerAdd : int[int, int];
    funcPointerSubtract : int[int, int];
    funcPointerDivide : int[int, int];
    funcPointerMultiply : int[int, int];
}

function main : int ()
{
    space : container;
    funcPointerContainer : @container;
    funcPointerContainer = &space;

    result1 : int;
    result2 : int;
    result3 : int;
    result4 : int;

    funcPointerContainer->funcPointerAdd = &add;
    funcPointerContainer->funcPointerSubtract = &subtract;
    funcPointerContainer->funcPointerDivide = &divide;
    funcPointerContainer->funcPointerMultiply = &multiply;

    result1 = funcPointerContainer->funcPointerAdd(18, 6);
    result2 = funcPointerContainer->funcPointerSubtract(18, 6);
    result3 = funcPointerContainer->funcPointerDivide(18, 6);
    result4 = funcPointerContainer->funcPointerMultiply(18, 6);

    yesOrNo : int;
    if (result1 == 24 && result2 == 12 && result3 == 3 && result4 == 108)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function add : int (n : int, m : int) 
{
    return n + m;
}

function subtract : int (n : int, m : int) 
{
    return n - m;
}

function divide : int (n : int, m : int)
{
    return n / m;
}

function multiply : int (n : int, m : int)
{
    return n * m;
}

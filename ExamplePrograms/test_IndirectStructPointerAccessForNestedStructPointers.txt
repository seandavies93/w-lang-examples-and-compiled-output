struct example
{
    first : int;
    second : int;
    third : int;
}

struct example2
{
    point : @example;
}

function main : int ()
{
    var1 : example2;

    var2 : example;
    var2.first = 1;
    var2.second = 2;
    var2.third = 3;

    var1.point = &var2;

    pointer : @example2;
    pointer = &var1;
    
    result : int;
    yesOrNo : int;

    result = pointer->point->first;

    if (result == 1)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

struct exponentialData
{
    forTwo : baseTwo;
    forThree : baseThree;
}

struct baseTwo
{
    toTheFour : int;
    toTheFive : int;
    toTheSix : int;
}

struct baseThree
{
    toTheFour : int;
    toTheFive : int;
    toTheSix : int;
}

function main : int ()
{
    data : exponentialData;
    data = populateExponentialData();

    result : int;
    result = checkDataCorrect(data);

    yesOrNo : int;
    if (result)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function populateExponentialData : exponentialData ()
{
    data : exponentialData;
    data.forTwo.toTheFour = exponentiate(2, 4);
    data.forTwo.toTheFive = exponentiate(2, 5);
    data.forTwo.toTheSix = exponentiate(2, 6);
    data.forThree.toTheFour = exponentiate(3, 4);
    data.forThree.toTheFive = exponentiate(3, 5);
    data.forThree.toTheSix = exponentiate(3, 6);
    return data;
}

function checkDataCorrect : int (data : exponentialData)
{
    return checkForTwo(data.forTwo) && checkForThree(data.forThree);
}

function checkForTwo : int (two : baseTwo)
{
    return two.toTheFour == 16 && two.toTheFive == 32 && two.toTheSix == 64;
}

function checkForThree : int (three : baseThree)
{
    return three.toTheFour == 81 && three.toTheFive == 243 && three.toTheSix == 729;
}

function exponentiate : int (base : int, exponent : int)
{
    cumulant : int;
    cumulant = 1;
    index : int;
    index = 1;
    
    while (index <= exponent)
    {
        cumulant = cumulant * base;
        index = index + 1;
    }
    return cumulant;
}

function main : int ()
{
    result1 : int;
    result2 : int;
    result3 : int;
    result4 : int;
    result5 : int;
    result6 : int;
    result1 = recurrence(0);
    result2 = recurrence(1);
    result3 = recurrence(2);
    result4 = recurrence(3);
    result5 = recurrence(4);
    result6 = recurrence(5);

    yesOrNo : int;
    if (result1 == 1 && result2 == 2 && result3 == 7 && result4 == 20 && result5 == 61 && result6 == 182)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function recurrence : int (n : int)
{
    if (n == 0)
    {
        return 1;
    } 
    else if (n == 1)
    {
        return 2;
    }
    return 2 * recurrence(n-1) + 3 * recurrence(n-2);
}

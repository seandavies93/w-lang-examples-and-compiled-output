function main : int ()
{
    array : @int;
    array = initIntArray();

    yesOrNo : int;
    if (array[0] == 1 &&
        array[1] == 2 &&
        array[2] == 3)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function initIntArray : @int ()
{
    firstAllocation : @int;
    firstAllocation = halloc(3);
    firstAllocation[0] = 1;
    firstAllocation[1] = 2;
    firstAllocation[2] = 3;
    return firstAllocation;
}

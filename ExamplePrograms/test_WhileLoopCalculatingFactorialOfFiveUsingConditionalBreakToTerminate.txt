function main : int ()
{
    result : int;
    result = 1;

    index : int;
    index = 1;

    valueForFactorial : int;
    valueForFactorial = 5;

    while (index >= 0)
    {
        result = result * index;

        if (index == valueForFactorial)
        {
            break;
        }

        index = index + 1;
    }

    yesOrNo : int;
    if (result == 120)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

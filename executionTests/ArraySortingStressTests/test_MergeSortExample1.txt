function main : int ()
{
    array : int{10};

    array[0] = 4;
    array[1] = 6;
    array[2] = 5;
    array[3] = 7;
    array[4] = 9;
    array[5] = 1;
    array[6] = 8;
    array[7] = 10;
    array[8] = 3;
    array[9] = 2;

    result : @int;

    result = mergeSort(array, 10);

    yesOrNo : int;
    if (
        result[0] == 1 &&
        result[1] == 2 &&
        result[2] == 3 &&
        result[3] == 4 &&
        result[4] == 5 &&
        result[5] == 6 &&
        result[6] == 7 &&
        result[7] == 8 &&
        result[8] == 9 &&
        result[9] == 10)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    free(result);
    return yesOrNo;
}

function mergeSort : @int (array : @int, size : int)
{
    if (size == 1)
    {
        single : @int;
        single = halloc(1);
        @single = array[0];
        return single;
    }

    midpoint : int;
    midpoint = size / 2;

    firstSubArraySize : int;
    secondSubArraySize : int;

    firstSubArraySize = midpoint;
    secondSubArraySize = size - midpoint;

    first : @int;
    second : @int;

    first = mergeSort(array, firstSubArraySize);
    second = mergeSort(&array[midpoint], secondSubArraySize);

    target : @int;
    target = halloc(size);

    i : int;
    j : int;
    t : int;
    i = 0;
    j = 0;
    t = 0;

    while (i < firstSubArraySize && j < secondSubArraySize)
    {
        if (first[i] < second[j])
        {
            target[t] = first[i];
            i = i + 1;
        }
        else
        {
            target[t] = second[j];
            j = j + 1;
        }
        t = t + 1;
    }

    while (i < firstSubArraySize)
    {
        target[t] = first[i];
        i = i + 1;
        t = t + 1;
    }

    while (j < secondSubArraySize)
    {
        target[t] = second[j];
        j = j + 1;
        t = t + 1;
    }

    free(first);
    free(second);
    return target;
}

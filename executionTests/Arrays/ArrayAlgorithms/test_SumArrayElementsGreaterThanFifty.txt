function sumArrayElementsGreaterThanFifty : int (array : @int, size : int)
{
    index : int;
    index = 0;
    result : int;
    result = 0;

    while (index < size)
    {
        if (array[index] > 50)
        {
            result = result + array[index];
        }
        index = index + 1;
    }
    return result;
}

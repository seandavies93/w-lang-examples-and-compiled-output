struct example
{
    zeroth : int;
    first : int;
    second : int;
    third : int;
    fourth : int;
    pointer : @int;
}

struct example2
{
    point : example;
    point1 : example;
}

struct example3
{
    item : @example2;
    item1 : @example2;
    item2 : @example2;
}

struct example4
{
    item : example3;
    item1 : example3;
    item2 : example3;
}

struct example5
{
    item : example4;
    item1 : example4;
    item2 : example4;
}


function main : int ()
{
    var2 : example2;
    item1 : int;
    item2 : int;
    item3 : int;
    item4 : int;
    item5 : int;
    var5 : example5;
    item1 = 1;
    item2 = 2;
    item3 = 3;
    item4 = 4;
    item5 = 5;
    toPopulate : @example5;
    
    toPopulate = &var5;
    var5.item2.item.item1 = &var2;

    populate(toPopulate);

    var5.item2.item.item1->point.pointer = &item1;

    result1 : int;
    result2 : int;
    result3 : int;
    result4 : int;
    result5 : int;
    yesOrNo : int;

    result1 = var5.item2.item.item1->point.pointer[var5.item2.item.item1->point1.zeroth];
    result2 = var5.item2.item.item1->point.pointer[var5.item2.item.item1->point1.first];
    result3 = var5.item2.item.item1->point.pointer[var5.item2.item.item1->point1.second];
    result4 = var5.item2.item.item1->point.pointer[var5.item2.item.item1->point1.third];
    result5 = var5.item2.item.item1->point.pointer[var5.item2.item.item1->point1.fourth];

    if (result1 == 1 && result2 == 2 && result3 == 3 && result4 == 4 && result5 == 5)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}


function populate : void(toPopulate : @example5)
{
    toPopulate->item2.item.item1->point1.zeroth = 0;
    toPopulate->item2.item.item1->point1.first = 1;
    toPopulate->item2.item.item1->point1.second = 2;
    toPopulate->item2.item.item1->point1.third = 3;
    toPopulate->item2.item.item1->point1.fourth = 4;
}

function main : int ()
{
    array1 : int;
    array2 : int;
    array3 : int;
    array4 : int;

    initialiseArray(&array1);

    yesOrNo : int;
    if (array1 == 1 && array2 == 2 && array3 == 6 && array4 == 24)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function initialiseArray : void (array : @int)
{
    index : int;
    index = 0;

    while (index < 4)
    {
        array[index] = factorial(index + 1);
        index = index + 1;
    }
}

function factorial : int (n : int)
{
    value : int;
    if (n > 0)
    {
        value = n * factorial(n-1);
    }
    else
    {
        value = 1;
    }

    return value;
}

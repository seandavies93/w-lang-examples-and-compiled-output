function main : int ()
{
    rows : int;
    cols : int;
    rows = 3;
    cols = 3;

    matrix : @@int;
    matrix = halloc(rows);

    i : int;
    i = 0;
    while (i < rows)
    {
        matrix[i] = halloc(cols);
        i = i + 1;
    }

    matrix[0][0] = 5;
    matrix[0][1] = 12;
    matrix[0][2] = 8;

    matrix[1][0] = 3;
    matrix[1][1] = 1;
    matrix[1][2] = 15;

    matrix[2][0] = 7;
    matrix[2][1] = 4;
    matrix[2][2] = 10;

    totalSum : int;
    totalSum = sumEntireMatrix(matrix, rows, cols);


    i = 0;
    while (i < rows)
    {
        free(matrix[i]);
        i = i + 1;
    }
    free(matrix);

    yesOrNo : int;
    if (totalSum == 65)
    {
        yesOrNo = 89; 
    }
    else
    {
        yesOrNo = 78;
    }

    return yesOrNo;
}

function sumEntireMatrix : int (matrix : @@int, rows : int, cols : int)
{
    accumulator : int;
    accumulator = 0;

    r : int;
    c : int;
    r = 0;

    while (r < rows)
    {
        c = 0;
        while (c < cols)
        {
            accumulator = accumulator + matrix[r][c];
            c = c + 1;
        }
        r = r + 1;
    }

    return accumulator;
}

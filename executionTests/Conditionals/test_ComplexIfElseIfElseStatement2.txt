function main : int ()
{
    result0 : int;
    result1 : int;
    result2 : int;
    result3 : int;
    result4 : int;
    result5 : int;
    resultMinus1 : int;
    
    minus1 : int;
    minus1 = 0 - 1;

    resultMinus1 = computeSelective(minus1);
    result0 = computeSelective(0);
    result1 = computeSelective(1);
    result2 = computeSelective(2);
    result3 = computeSelective(3);
    result4 = computeSelective(4);
    result5 = computeSelective(5);

    yesOrNo : int;
    if (resultMinus1 == 137 && result0 == 83 && result1 == 14 && result2 == 18 && result3 == 21 && result4 == 43 && result5 == 67)
    {
        yesOrNo = 89;

}

function computeSelective : int (choice : int)
{
    result : int;   
    if (choice <= 1)
    {
        if (choice == 1)
        {
            result = 14;
        }
        else if (choice == 0)
        {
            result = 83;
        }
        else
        {
            result = 137;
        }
    }
    else if (choice == 2)
    {
        result = 18;
    }
    else if (choice == 3)
    {
        result = 21;
    }
    else if (choice >= 4)
    {
        if (choice == 4)
        {
            result = 43;
        }
        else
        {
            result = 67;
        }
    }
    
    return result;
}

function main : int ()
{
    outer : int;
    inner : int;
    total : int;

    outer = 0;
    total = 0;

    while (outer < 5)
    {
        inner = 0;
        outer = outer + 1;

        while (inner < 5)
        {
            inner = inner + 1;

            if (inner % 2 == 0)
            {
                continue;
            }

            total = total + outer;
        }

        if (outer == 4)
        {
            break;
        }
    }

    yesOrNo : int;
    if (total == 30)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

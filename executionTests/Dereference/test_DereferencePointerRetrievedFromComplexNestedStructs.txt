struct example3

    struct1 : example1;
    struct2 : example2;
    struct3 : example3;

    struct2.item1 = &struct1;
    struct1.pointer = &item;

    structPointer : @example3;
    structPointer = &struct3;

    pointInt : @int;
    pointInt = structPointer->item3->item1->pointer;

    @pointInt = 21;

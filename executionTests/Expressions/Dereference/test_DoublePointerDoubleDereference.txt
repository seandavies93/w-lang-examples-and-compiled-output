function main : int ()
{
    value : int;
    pointer : @int;
    doublePointer : @@int;

    value = 11;
    pointer = &value;
    doublePointer = &pointer;

    yesOrNo : int;
    if (@@doublePointer == 11)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

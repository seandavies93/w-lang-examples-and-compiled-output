struct PointerToInt
{
    pointer : @int;
}

function main : int ()
{
    value1 : int;
    value2 : int;
    pointer1 : @int;
    pointer2 : @int;
    
    pointerToPointer1 : @PointerToInt;
    pointerToPointer2 : @PointerToInt;

    value1 = 5;
    value2 = 10;

    pointer1 = &value1;
    pointer2 = &value2;

    pointerToPointer1 = &pointer1;
    pointerToPointer2 = &pointer2;

    yesOrNo : int;
    if (@@pointerToPointer1 == 5 && @@pointerToPointer2 == 10)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

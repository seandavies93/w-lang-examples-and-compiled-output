function main : int ()
{
    result1 : int;
    result2 : int;
    result3 : int;
    result4 : int;

    result1 = callFunctionPointer(&add, 18, 6);
    result2 = callFunctionPointer(&subtract, 18, 6);
    result3 = callFunctionPointer(&divide, 18, 6);
    result4 = callFunctionPointer(&multiply, 18, 6);

    yesOrNo : int;
    if (result1 == 24 && result2 == 12 && result3 == 3 && result4 == 108)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function add : int (n : int, m : int) 
{
    return n + m;
}

function subtract : int (n : int, m : int) 
{
    return n - m;
}

function divide : int (n : int, m : int)
{
    return n / m;
}

function multiply : int (n : int, m : int)
{
    return n * m;
}

function callFunctionPointer : int (funcPointer : int[int, int], n : int, m : int)
{
    return funcPointer(n, m);
}

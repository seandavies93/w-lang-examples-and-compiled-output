struct example
{
    first : int;
    second : int;
    third : int;
}

struct example1
{
    funcPointer : @example[];
}

function main : int ()
{
    retVal : @example;
    item : example;
    result : int;
    funcPointerContainer : example1;
    funcPointerContainer.funcPointer = &coolFunction;
    retVal = funcPointerContainer.funcPointer();

    item = @retVal;

    result = item.third;

    yesOrNo : int;
    if (result == 3)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function coolFunction : @example ()
{
    next : example;
    next.first = 1;
    next.second = 2;
    next.third = 3;
    return &next;
}

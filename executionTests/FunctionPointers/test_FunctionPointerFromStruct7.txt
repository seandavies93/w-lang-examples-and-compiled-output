struct example
{
    value : int;
    index : int;
}

struct example1
{
    funcPointer : void[int, @example];
}

function main : int ()
{
    struct1 : example;
    funcPointerContainer : example1;
    funcPointerContainer.funcPointer = &factorial;
    funcPointerContainer.funcPointer(5, &struct1);

    yesOrNo : int;
    if (struct1.value == 120)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function factorial : void (n : int, workArea : @example)
{
    workArea->value = 1;
    workArea->index = 1;
    
    while (workArea->index <= n)
    {
        workArea->value = workArea->value * workArea->index;
        workArea->index = workArea->index + 1;
    }
}

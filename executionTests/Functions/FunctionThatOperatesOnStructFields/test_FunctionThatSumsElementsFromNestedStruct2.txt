struct example
{
    a : int;
    b : int;
}

struct example1
{
    first : example;
    second : example;
}

function main : int ()
{
    value1 : example;
    value1.a = 1;
    value1.b = 2;

    value2 : example;
    value2.a = 3;
    value2.b = 4;

    package : example1;
    package.first = value1;
    package.second = value2;

    result : int;

    result = sum(package);

    yesOrNo : int;
    if (result == 10)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function sum : int (package : example1)
{
    return package.first.a + package.first.b + package.second.a + package.second.b;
}

function isPrime : int (number : int) 
{
    if (number >= 2)
    {
        divisor : int;
        divisor = 2;

        while (divisor < number)
        {
            if (((number / divisor) * divisor) == number)
            {
                return 0;
            }
            divisor = divisor + 1;
        }
        return 1;
    }
    else
    {
        return 0;
    }
}

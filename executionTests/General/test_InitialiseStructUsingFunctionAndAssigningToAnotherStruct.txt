struct example
{
    first : int;
    second : int;
    third : int;
}

function main : int ()
{
    other : example;
    retVal : example;
    yesOrNo : int;

    retVal = coolFunction();

    other = retVal;
    
    if (other.first == 1 && other.second == 2 && other.third == 3)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function coolFunction : example ()
{
    next : example;
    next.first = 1;
    next.second = 2;
    next.third = 3;
    return next;
}

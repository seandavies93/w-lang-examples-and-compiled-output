struct PointerToInt
{
    pointer : @int;
}

function main : int ()
{
    value1 : int;
    value2 : int;
    pointer1 : @int;
    pointer2 : @int;
    beforeSwapCorrect : int;
    
    pointerToPointer1 : @PointerToInt;
    pointerToPointer2 : @PointerToInt;

    value1 = 5;
    value2 = 10;

    pointer1 = &value1;
    pointer2 = &value2;

    beforeSwapCorrect = @pointer1 == 5 && @pointer2 == 10;

    pointerToPointer1 = &pointer1;
    pointerToPointer2 = &pointer2;

    

    temp: @int;
    temp = pointerToPointer2->pointer;
    pointerToPointer2->pointer = pointerToPointer1->pointer;
    pointerToPointer1->pointer = temp;

    yesOrNo : int;
    if (beforeSwapCorrect && value1 == 5 && value2 == 10 && @pointer1 == 10 && @pointer2 == 5)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

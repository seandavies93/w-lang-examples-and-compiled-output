struct example
{
    contained : example2;
}

struct example2
{
    a : int;
    b : int;
}

function main : int ()
{
    result : example2;
    result = coolFunction().contained;

    yesOrNo : int;
    if (result.a == 1 && result.b == 2)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function coolFunction : example ()
{
    dep : example2;
    dep.a = 1;
    dep.b = 2;

    next : example;
    next.contained = dep;

    return next;
}

struct example1
{
    item : @example2;
}

struct example2
{
    nestedItem : example3;
}

struct example3
{
    a : int;
    b : int;
}

function main : int ()
{
    item2 : @example2;
    item2 = halloc(2);

    item2->nestedItem.a = 10;
    item2->nestedItem.b = 20;

    result : example3;
    result = coolFunction(item2).item->nestedItem;

    free(item2);

    yesOrNo : int;
    if (result.a == 10 && result.b == 20)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function coolFunction : example1 (itemPointer : @example2)
{
    item1 : example1;
    item1.item = itemPointer;

    return item1;
}

struct dataForComputation
{
    input : int;
    output : int;
}

function main : int ()
{
    dataArray : dataForComputation{3};

    dataArray[0].input = 4;
    dataArray[1].input = 5;
    dataArray[2].input = 6;

    dataArray[0] = multiplyByFour(dataArray[0]);
    dataArray[1] = multiplyByFour(dataArray[1]);
    dataArray[2] = multiplyByFour(dataArray[2]);

    yesOrNo : int;

    if (dataArray[0].output == 16 && dataArray[1].output == 20 && dataArray[2].output == 24)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}

function multiplyByFour : dataForComputation (data : dataForComputation)
{
    index : int;
    index = 1;

    data.output = 0;
    
    while (index <= 4)
    {
        data.output = data.output + data.input;
        index = index + 1;
    }
    return data;
}

struct example
{
    first : int;
    second : int;
    third : int;
}

struct example1
{
    contained : example;
    contained1 : example;
}

function main : int ()
{
    targetStruct : example1;
    pointer : @example1;
    pointer = &targetStruct;

    pointer->contained1.first = 1;
    pointer->contained1.second = 2;
    pointer->contained1.third = 3;
    
    result1 : int;
    result2 : int;
    result3 : int;
    
    result1 = (pointer->contained1).first;
    result2 = (pointer->contained1).second;
    result3 = (pointer->contained1).third;

    yesOrNo : int;
    if (
        result1 == 1 && 
        result2 == 2 && 
        result3 == 3)
    {
        yesOrNo = 89;
    }
    else
    {
        yesOrNo = 78;
    }
    return yesOrNo;
}
